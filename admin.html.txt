  <script>
    const TASKS = [
      { id: 't1', title: '借閱 3 本書', icon: 'fa-book-open' },
      { id: 't2', title: '參加一場講座/活動', icon: 'fa-users' },
      { id: 't3', title: '尋找展覽隱藏密碼', icon: 'fa-magnifying-glass' },
      { id: 't4', title: '填寫活動問卷', icon: 'fa-pen-to-square' }
    ];

    let currentAdminCardId = null;

    function getDB() {
      return JSON.parse(localStorage.getItem('lib_card_db')) || {};
    }

    function saveDB(db) {
      localStorage.setItem('lib_card_db', JSON.stringify(db));
    }

    function adminLoadCard(cardIdOverride) {
      const cardId = (cardIdOverride || document.getElementById('admin-card-input').value).trim().toUpperCase();
      if (!cardId) return alert('請先輸入或掃描卡號');

      currentAdminCardId = cardId;
      const db = getDB();
      if (!db[cardId]) {
        db[cardId] = { completed: [], claimed: false };
        saveDB(db);
      }

      renderAdminPanel();
    }

    function renderAdminPanel() {
      const db = getDB();
      const data = db[currentAdminCardId];
      document.getElementById('admin-display-card-id').textContent = currentAdminCardId;

      const claimArea = document.getElementById('admin-claim-area');
      const isAllDone = data.completed.length === TASKS.length;

      if (data.claimed) {
        claimArea.innerHTML = `<span class="text-slate-400 text-xs font-bold bg-slate-100 px-3 py-1.5 rounded-xl border border-slate-200">獎品已發放</span>`;
      } else if (isAllDone) {
        claimArea.innerHTML = `<button onclick="claimReward()" class="bg-emerald-600 hover:bg-emerald-500 text-white font-bold px-4 py-2 rounded-xl shadow-md transition animate-bounce text-xs"><i class="fa-solid fa-gift mr-1"></i> 發放兌換獎品</button>`;
      } else {
        claimArea.innerHTML = `<span class="text-amber-600 text-xs font-bold bg-amber-50 px-3 py-1.5 rounded-lg border border-amber-200">尚未集滿</span>`;
      }

      const adminTaskList = document.getElementById('admin-task-list');
      adminTaskList.innerHTML = TASKS.map(task => {
        const isDone = data.completed.includes(task.id);
        return `
          <div class="flex justify-between items-center p-3 rounded-xl border ${isDone ? 'bg-indigo-50 border-indigo-200' : 'bg-white border-slate-200'}">
            <div class="flex items-center gap-3">
              <i class="fa-solid ${task.icon} ${isDone ? 'text-indigo-600' : 'text-slate-400'} text-base"></i>
              <span class="font-bold text-sm ${isDone ? 'text-indigo-900' : 'text-slate-700'}">${task.title}</span>
            </div>
            <button onclick="toggleTaskStatus('${task.id}')" class="px-3.5 py-1.5 rounded-lg text-xs font-bold transition ${isDone ? 'bg-indigo-600 text-white' : 'bg-slate-100 text-slate-600 hover:bg-slate-200'}">
              ${isDone ? '<i class="fa-solid fa-circle-check mr-1"></i> 已完成' : '蓋章解鎖'}
            </button>
          </div>
        `;
      }).join('');

      document.getElementById('admin-panel').classList.remove('hidden');
    }

    function toggleTaskStatus(taskId) {
      const db = getDB();
      const data = db[currentAdminCardId];
      const index = data.completed.indexOf(taskId);

      if (index > -1) {
        data.completed.splice(index, 1);
      } else {
        data.completed.push(taskId);
        confetti({ particleCount: 50, spread: 60, origin: { y: 0.8 } });
      }

      saveDB(db);
      renderAdminPanel();
    }

    function claimReward() {
      if (confirm(`確定要為卡號 ${currentAdminCardId} 發放獎品嗎？`)) {
        const db = getDB();
        db[currentAdminCardId].claimed = true;
        saveDB(db);
        confetti({ particleCount: 150, spread: 100, origin: { y: 0.6 } });
        renderAdminPanel();
      }
    }

    let html5QrcodeScanner = null;
    function toggleScanner() {
      const container = document.getElementById('scanner-container');
      const isHidden = container.classList.contains('hidden');

      if (isHidden) {
        container.classList.remove('hidden');
        document.getElementById('scanner-toggle-btn').innerHTML = `<i class="fa-solid fa-xmark"></i> 關閉鏡頭`;
        html5QrcodeScanner = new Html5QrcodeScanner("interactive-reader", { fps: 10, qrbox: 250 });
        html5QrcodeScanner.render((decodedText) => {
          document.getElementById('admin-card-input').value = decodedText;
          adminLoadCard(decodedText);
          toggleScanner();
        });
      } else {
        container.classList.add('hidden');
        document.getElementById('scanner-toggle-btn').innerHTML = `<i class="fa-solid fa-camera"></i> 開啟鏡頭掃描`;
        if (html5QrcodeScanner) html5QrcodeScanner.clear();
      }
    }
  </script>
