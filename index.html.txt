  <script>
    const TASKS = [
      { id: 't1', title: '借閱 3 本書', desc: '至自助借書機或櫃台借閱 3 本圖書', icon: 'fa-book-open' },
      { id: 't2', title: '參加一場講座/活動', desc: '參與圖書館舉辦的實體主題活動', icon: 'fa-users' },
      { id: 't3', title: '尋找展覽隱藏密碼', desc: '完成主題展覽區的解謎學習單', icon: 'fa-magnifying-glass' },
      { id: 't4', title: '填寫活動問卷', desc: '線上或紙本完成本季滿意度問卷', icon: 'fa-pen-to-square' }
    ];

    function searchReaderCard() {
      const cardId = document.getElementById('reader-card-input').value.trim().toUpperCase();
      if (!cardId) return alert('請輸入卡號/條碼號！');

      const db = JSON.parse(localStorage.getItem('lib_card_db')) || {};
      const data = db[cardId] || { completed: [], claimed: false };

      document.getElementById('display-card-id').textContent = cardId;
      
      const count = data.completed.length;
      const total = TASKS.length;
      const percent = (count / total) * 100;

      document.getElementById('progress-text').textContent = `${count} / ${total}`;
      document.getElementById('progress-bar').style.width = `${percent}%`;

      const rewardBadge = document.getElementById('reward-status-badge');
      if (data.claimed) {
        rewardBadge.innerHTML = `<span class="bg-slate-100 text-slate-600 font-bold px-3 py-1 rounded-xl border border-slate-200 text-xs"><i class="fa-solid fa-gift text-slate-400 mr-1"></i> 已領取獎品</span>`;
      } else if (count === total) {
        rewardBadge.innerHTML = `<span class="bg-emerald-100 text-emerald-700 font-bold px-3 py-1 rounded-xl border border-emerald-200 text-xs animate-pulse"><i class="fa-solid fa-gift text-emerald-600 mr-1"></i> 可至櫃檯領獎！</span>`;
      } else {
        rewardBadge.innerHTML = `<span class="bg-amber-50 text-amber-600 font-bold px-3 py-1 rounded-xl border border-amber-200 text-xs"><i class="fa-solid fa-hourglass-half mr-1"></i> 還差 ${total - count} 個任務</span>`;
      }

      const taskGrid = document.getElementById('reader-task-grid');
      taskGrid.innerHTML = TASKS.map(task => {
        const isDone = data.completed.includes(task.id);
        return `
          <div class="flex items-center p-3.5 rounded-xl border ${isDone ? 'bg-indigo-50/60 border-indigo-200' : 'bg-slate-50 border-slate-200 badge-locked'}">
            <div class="w-10 h-10 rounded-xl flex items-center justify-center text-lg font-bold ${isDone ? 'bg-indigo-600 text-white' : 'bg-slate-200 text-slate-400'}">
              <i class="fa-solid ${isDone ? 'fa-check' : task.icon}"></i>
            </div>
            <div class="ml-3.5">
              <h5 class="font-bold text-sm ${isDone ? 'text-indigo-900' : 'text-slate-600'}">${task.title}</h5>
              <p class="text-xs text-slate-500 mt-0.5">${task.desc}</p>
            </div>
          </div>
        `;
      }).join('');

      document.getElementById('reader-status-card').classList.remove('hidden');
    }
  </script>
